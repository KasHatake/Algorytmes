#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <time.h>
#include <windows.h> 
#define ROW 10
#define COL 10
#define _CRT_SECURE_NO_WARNINGS
#define MAX_SIZE 100  
#define NAME_LENGTH 20

struct Student {
    char first_name[NAME_LENGTH];
    char last_name[NAME_LENGTH];
};

struct Node {
    struct Student student;
    struct Node* next;
};

struct Node* createNode(struct Student student) {
    struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));
    if (newNode == NULL) {
        printf("Ошибка выделения памяти!\n");
        return NULL;
    }
    newNode->student = student;
    newNode->next = NULL;
    return newNode;
}

void addStudent(struct Node** head, struct Student student, int showMessage) {
    struct Node* newNode = createNode(student);
    if (*head == NULL) {
        *head = newNode;
        if (showMessage) {
            printf("Студент %s %s добавлен.\n", student.first_name, student.last_name);
        }
        return;
    }

    struct Node* current = *head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = newNode;
    if (showMessage) {
        printf("Студент %s %s добавлен.\n", student.first_name, student.last_name);
    }
}

void deleteStudentByIndex(struct Node** head, int index) {
    if (*head == NULL) {
        printf("Список пуст!\n");
        return;
    }

    if (index < 1) {
        printf("Неверный индекс! Индекс должен быть больше 0.\n");
        return;
    }

    struct Node* current = *head;
    struct Node* prev = NULL;
    int currentIndex = 1;

    while (current != NULL && currentIndex < index) {
        prev = current;
        current = current->next;
        currentIndex++;
    }

    if (current == NULL) {
        printf("Студент с индексом %d не найден.\n", index);
        return;
    }

    if (prev == NULL) {
        *head = current->next;
    }
    else {
        prev->next = current->next;
    }

    printf("Студент %s %s (индекс %d) удален.\n",
        current->student.first_name, current->student.last_name, index);
    free(current);
}

void printList(struct Node* head) {
    if (head == NULL) {
        printf("Список пуст!\n");
        return;
    }

    int count = 1;
    struct Node* current = head;
    printf("\nСписок студентов:\n");
    printf("================\n");

    while (current != NULL) {
        printf("%d. %s %s\n", count++, current->student.first_name, current->student.last_name);
        current = current->next;
    }
    printf("================\n");
}

void searchInList(struct Node* head, char* search) {
    struct Node* current = head;
    int found = 0;

    printf("\nРезультаты поиска для '%s':\n", search);

    while (current != NULL) {
        if (strcmp(current->student.first_name, search) == 0 ||
            strcmp(current->student.last_name, search) == 0) {
            printf("- %s %s\n", current->student.first_name, current->student.last_name);
            found++;
        }
        current = current->next;
    }

    if (found == 0) {
        printf("Студенты не найдены.\n");
    }
    else {
        printf("Найдено: %d студент(ов)\n", found);
    }
}

void freeList(struct Node* head) {
    struct Node* current = head;
    while (current != NULL) {
        struct Node* temp = current;
        current = current->next;
        free(temp);
    }
}

struct Student inputStudent() {
    struct Student student;
    printf("Введите имя: ");
    scanf_s("%s", student.first_name, NAME_LENGTH);
    printf("Введите фамилию: ");
    scanf_s("%s", student.last_name, NAME_LENGTH);
    return student;
}
